\chapter{Amplitudes de diffusion Raman doublement résonante}\label{annexe:Raman}
Nous donnons les expressions explicites des amplitudes de diffusion Raman doublement résonante $K^{\text{pp}}(\kv,\qv,\nu,\mu)$ et $K^{\text{pd}}(\kv,\qv,\nu)$ qui entrent dans l'évaluation de l'intensité Raman au deuxième ordre via les équations~\eqref{eqn:Raman_2nd}--\eqref{eqn:I_pd}. Pour que les dénominateurs des amplitudes tiennent sur une ligne, nous changeons légèrement la notation utilisée : $\tilde{\varepsilon}_{n\kv}=\tilde{\varepsilon}_\kv^n$, $\omega_{\nu\qv}=\omega_\qv^\nu$ et nous définissons $\varepsilon_L=\hbar\omega_L$.
\section{Processus \textit{deux-phonons}}
Il y a huit processus \textit{deux-phonons}~\cite{venezuelaTheoryDoubleresonantRaman2011}.

\noindent
Processus \textit{ee1}, où l'électron est diffusé par le phonon $-\nu\qv$, puis par le phonon $\mu\qv$.
\begin{equation}
        K_{ee1}^{\text{pp}} = \frac{g^{(\lambda_S)}_{\pi\pi^*}(\kv)g^{(\mu)}_{\pi^*\pi^*}(\kv+\qv,-\qv)g_{\pi^*\pi^*}^{(\nu)}(\kv,\qv)g_{\pi^*\pi}^{(\lambda_L)}(\kv)}{(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_\kv^\pi-\hbar\omega_{-\qv}^\nu-\hbar\omega_\qv^\mu-i\frac{\Gamma_c}{2})(\varepsilon_L-\tilde{\varepsilon}_{\kv+\qv}^{\pi^*}+\tilde{\varepsilon}_\kv^\pi-\hbar\omega_{-\qv}^\nu-i\frac{\Gamma_b}{2})(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_{\kv}^\pi-i\frac{\Gamma_a}{2})}\label{eqn:pp_ee1}.
\end{equation}
Processus \textit{ee2}, où l'électron est diffusé par le phonon $\mu\qv$, puis par le phonon $-\nu\qv$.
\begin{equation}
        K_{ee2}^{\text{pp}} = \frac{g^{(\lambda_S)}_{\pi\pi^*}(\kv)g^{(\nu)}_{\pi^*\pi^*}(\kv-\qv,\qv)g_{\pi^*\pi^*}^{(\mu)}(\kv,-\qv)g_{\pi^*\pi}^{(\lambda_L)}(\kv)}{(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_\kv^\pi-\hbar\omega_{-\qv}^\nu-\hbar\omega_\qv^\mu-i\frac{\Gamma_c}{2})(\varepsilon_L-\tilde{\varepsilon}_{\kv-\qv}^{\pi^*}+\tilde{\varepsilon}_\kv^\pi-\hbar\omega_{\qv}^\mu-i\frac{\Gamma_b}{2})(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_{\kv}^\pi-i\frac{\Gamma_a}{2})}\label{eqn:pp_ee2}.
\end{equation}
Processus \textit{hh1}, où le trou est diffusé par le phonon $-\nu\qv$, puis par le phonon $\mu\qv$.
\begin{equation}
        K_{hh1}^{\text{pp}} = \frac{g^{(\lambda_S)}_{\pi\pi^*}(\kv)g^{(\mu)}_{\pi\pi}(\kv,-\qv)g_{\pi\pi}^{(\nu)}(\kv-\qv,\qv)g_{\pi^*\pi}^{(\lambda_L)}(\kv)}{(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_\kv^\pi-\hbar\omega_{-\qv}^\nu-\hbar\omega_\qv^\mu-i\frac{\Gamma_c}{2})(\varepsilon_L-\tilde{\varepsilon}_{\kv}^{\pi^*}+\tilde{\varepsilon}_{\kv-\qv}^\pi-\hbar\omega_{-\qv}^\nu-i\frac{\Gamma_b}{2})(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_{\kv}^\pi-i\frac{\Gamma_a}{2})}\label{eqn:pp_hh1}.
\end{equation}
Processus \textit{hh2}, où le trou est diffusé par le phonon $\mu\qv$, puis par le phonon $-\nu\qv$.
\begin{equation}
        K_{hh2}^{\text{pp}} = \frac{g^{(\lambda_S)}_{\pi\pi^*}(\kv)g^{(\nu)}_{\pi\pi}(\kv,\qv)g_{\pi\pi}^{(\mu)}(\kv+\qv,-\qv)g_{\pi^*\pi}^{(\lambda_L)}(\kv)}{(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_\kv^\pi-\hbar\omega_{-\qv}^\nu-\hbar\omega_\qv^\mu-i\frac{\Gamma_c}{2})(\varepsilon_L-\tilde{\varepsilon}_{\kv}^{\pi^*}+\tilde{\varepsilon}_{\kv+\qv}^\pi-\hbar\omega_{\qv}^\mu-i\frac{\Gamma_b}{2})(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_{\kv}^\pi-i\frac{\Gamma_a}{2})}\label{eqn:pp_hh2}.
\end{equation}
Processus \textit{eh1}, où l'électron est diffusé par le phonon $-\nu\qv$, puis le trou par le phonon $\mu\qv$.
\begin{equation}
        K_{eh1}^{\text{pp}} = \frac{g^{(\lambda_S)}_{\pi\pi^*}(\kv+\qv)g^{(\mu)}_{\pi\pi}(\kv+\qv,-\qv)g_{\pi^*\pi^*}^{(\nu)}(\kv,\qv)g_{\pi^*\pi}^{(\lambda_L)}(\kv)}{(\varepsilon_L-\tilde{\varepsilon}_{\kv+\qv}^{\pi^*}+\tilde{\varepsilon}_{\kv+\qv}^\pi-\hbar\omega_{-\qv}^\nu-\hbar\omega_\qv^\mu-i\frac{\Gamma_c}{2})(\varepsilon_L-\tilde{\varepsilon}_{\kv+\qv}^{\pi^*}+\tilde{\varepsilon}_{\kv}^\pi-\hbar\omega_{-\qv}^\nu-i\frac{\Gamma_b}{2})(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_{\kv}^\pi-i\frac{\Gamma_a}{2})}\label{eqn:pp_eh1}.
\end{equation}
Processus \textit{eh2}, où l'électron est diffusé par le phonon $\mu\qv$, puis le trou par le phonon $-\nu\qv$.
\begin{equation}
        K_{eh2}^{\text{pp}} = \frac{g^{(\lambda_S)}_{\pi\pi^*}(\kv-\qv)g^{(\nu)}_{\pi\pi}(\kv-\qv,\qv)g_{\pi^*\pi^*}^{(\mu)}(\kv,-\qv)g_{\pi^*\pi}^{(\lambda_L)}(\kv)}{(\varepsilon_L-\tilde{\varepsilon}_{\kv-\qv}^{\pi^*}+\tilde{\varepsilon}_{\kv-\qv}^\pi-\hbar\omega_{-\qv}^\nu-\hbar\omega_\qv^\mu-i\frac{\Gamma_c}{2})(\varepsilon_L-\tilde{\varepsilon}_{\kv-\qv}^{\pi^*}+\tilde{\varepsilon}_{\kv}^\pi-\hbar\omega_{\qv}^\mu-i\frac{\Gamma_b}{2})(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_{\kv}^\pi-i\frac{\Gamma_a}{2})}\label{eqn:pp_eh2}.
\end{equation}
Processus \textit{he1}, où le trou est diffusé par le phonon $-\nu\qv$, puis l'électron par le phonon $\mu\qv$.
\begin{equation}
        K_{he1}^{\text{pp}} = \frac{g^{(\lambda_S)}_{\pi\pi^*}(\kv-\qv)g^{(\mu)}_{\pi^*\pi^*}(\kv,-\qv)g_{\pi\pi}^{(\nu)}(\kv-\qv,\qv)g_{\pi^*\pi}^{(\lambda_L)}(\kv)}{(\varepsilon_L-\tilde{\varepsilon}_{\kv-\qv}^{\pi^*}+\tilde{\varepsilon}_{\kv-\qv}^\pi-\hbar\omega_{-\qv}^\nu-\hbar\omega_\qv^\mu-i\frac{\Gamma_c}{2})(\varepsilon_L-\tilde{\varepsilon}_{\kv}^{\pi^*}+\tilde{\varepsilon}_{\kv-\qv}^\pi-\hbar\omega_{-\qv}^\nu-i\frac{\Gamma_b}{2})(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_{\kv}^\pi-i\frac{\Gamma_a}{2})}\label{eqn:pp_he1}.
\end{equation}
Processus \textit{he2}, où le trou est diffusé par le phonon $\mu\qv$, puis l'électron par le phonon $-\nu\qv$.
\begin{equation}
        K_{he2}^{\text{pp}} = \frac{g^{(\lambda_S)}_{\pi\pi^*}(\kv+\qv)g^{(\nu)}_{\pi^*\pi^*}(\kv,\qv)g_{\pi\pi}^{(\mu)}(\kv+\qv,-\qv)g_{\pi^*\pi}^{(\lambda_L)}(\kv)}{(\varepsilon_L-\tilde{\varepsilon}_{\kv+\qv}^{\pi^*}+\tilde{\varepsilon}_{\kv+\qv}^\pi-\hbar\omega_{-\qv}^\nu-\hbar\omega_\qv^\mu-i\frac{\Gamma_c}{2})(\varepsilon_L-\tilde{\varepsilon}_{\kv}^{\pi^*}+\tilde{\varepsilon}_{\kv+\qv}^\pi-\hbar\omega_{\qv}^\mu-i\frac{\Gamma_b}{2})(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_{\kv}^\pi-i\frac{\Gamma_a}{2})}\label{eqn:pp_he2}.
\end{equation}

\section{Processus \textit{phonon-défaut}}
Il y a huit processus \textit{phonon-défaut}~\cite{venezuelaTheoryDoubleresonantRaman2011}.

\noindent
Processus \textit{ee1}, où l'électron est diffusé par le phonon $-\nu\qv$, puis par le défaut.
\begin{equation}
    K_{ee1}^{\text{pd}} = \frac{g^{(\lambda_S)}_{\pi\pi^*}(\kv)M_{\pi^*\pi^*}(\kv,\kv+\qv)g^{(\nu)}_{\pi^*\pi^*}(\kv,\qv)g_{\pi^*\pi}^{(\lambda_L)}(\kv)}{(\varepsilon_L-\tilde{\varepsilon}^{\pi^*}_\kv+\tilde{\varepsilon}^\pi_\kv-\hbar\omega_{-\qv}^\nu-i\frac{\Gamma_c}{2})(\varepsilon_L-\tilde{\varepsilon}_{\kv+\qv}^{\pi^*}+\tilde{\varepsilon}_\kv^\pi-\hbar\omega_{-\qv}^\nu-i\frac{\Gamma_b}{2})(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_\kv^\pi-i\frac{\Gamma_a}{2})}\label{eqn:pd_ee1}.
\end{equation}
Processus \textit{ee2}, où l'électron est diffusé par le défaut, puis par le phonon $-\nu\qv$.
\begin{equation}
        K_{ee2}^{\text{pd}} = \frac{g^{(\lambda_S)}_{\pi\pi^*}(\kv)g^{(\nu)}_{\pi^*\pi^*}(\kv-\qv,\qv)M_{\pi^*\pi^*}(\kv-\qv,\kv)g_{\pi^*\pi}^{(\lambda_L)}(\kv)}{(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_\kv^\pi-\hbar\omega_{-\qv}^\nu-i\frac{\Gamma_c}{2})(\varepsilon_L-\tilde{\varepsilon}_{\kv-\qv}^{\pi^*}+\tilde{\varepsilon}_\kv^\pi-i\frac{\Gamma_b}{2})(\varepsilon_L-\tilde{\varepsilon}_\kv^{\pi^*}+\tilde{\varepsilon}_{\kv}^\pi-i\frac{\Gamma_a}{2})}\label{eqn:pd_ee2}.
\end{equation}
